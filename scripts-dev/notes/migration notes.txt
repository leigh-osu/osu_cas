SELECT flps.entity_id AS nid, flac.field_lp_adj_column_value AS fc, flci.field_lp_col_image_fid AS fid
FROM field_data_field_larch_paragraph_sections flps
LEFT JOIN field_data_field_lp_adj_column flac
ON flps.field_larch_paragraph_sections_value = flac.entity_id
AND flps.field_larch_paragraph_sections_revision_id = flac.revision_id
RIGHT JOIN field_data_field_lp_col_image flci
ON flac.field_lp_adj_column_value = flci.entity_id
AND flac.field_lp_adj_column_revision_id = flci.revision_id
WHERE flps.entity_id IS NOT NULL;


WHERE flps.field_larch_paragraph_sections_value = 123587;
